Retrying in 5s... Error: {e}") time.sleep(5) return False def get_db_pool(): """Thread-safe on-demand pool allocator.""" global db_pool if db_pool is None: try: db_pool = pooling.MySQLConnectionPool( pool_name="inventory_pool", pool_size=10, host=DB_HOST, user=DB_USER, password=DB_PASSWORD, database=DB_NAME ) print("Connection pool created successfully.") except Error as e: print(f"Failed to create connection pool: {e}") raise e return db_pool def get_db(): """Attaches a pool connection to the application context 'g' for the current request.""" if 'db' not in g: pool = get_db_pool() g.db = pool.get_connection() return g.db @app.teardown_appcontext def close_db(error): """Automatically returns the connection to the pool at the end of the request.""" db = g.pop('db', None) if db is not None: db.close() # Get hidden items cursor.execute("SELECT * FROM inventory WHERE is_hidden = 1") hidden_items = cursor.fetchall() # Calculate summary arithmetic values cursor.execute("SELECT SUM(quantity) as total_qty, SUM(quantity * price) as total_val FROM inventory") stats = cursor.fetchone() cursor.close() total_quantity = stats['total_qty'] if stats['total_qty'] else 0 total_value = stats['total_val'] if stats['total_val'] else 0.0 return render_template( 'index.html', active_items=active_items, hidden_items=hidden_items, total_quantity=total_quantity, total_value=total_value ) except Error as e: flash(f"Error fetching data: {e}", "danger") return render_template('index.html', active_items=[], hidden_items=[], total_quantity=0, total_value=0.0) @app.route('/add', methods=['POST']) def add_item(): """Create a new inventory item.""" sku = request.form.get('sku') name = request.form.get('name') category = request.form.get('category') quantity = int(request.form.get('quantity', 0)) price = float(request.form.get('price', 0.0)) try: conn = get_db() cursor = conn.cursor() cursor.execute( "INSERT INTO inventory (sku, name, category, quantity, price) VALUES (%s, %s, %s, %s, %s)", (sku, name, category, quantity, price) ) conn.commit() cursor.close() flash(f"Item '{name}' added successfully! Only {current_qty} left in stock!", "danger") cursor.close() return redirect(url_for('index')) new_qty = current_qty - amount flash(f"Removed {amount} units from {item['name']}.", "warning") else: new_qty = current_qty cursor.execute("UPDATE inventory SET quantity = %s WHERE id = %s", (new_qty, item_id)) conn.commit() cursor.close() except Error as e: flash(f"Database error during stock adjustment: {e}", "danger") return redirect(url_for('index')) @app.route('/hide/', methods=['POST']) def hide_item(item_id): """Soft delete by archiving the item out of principal view.""" try: conn = get_db() cursor = conn.cursor() cursor.execute("UPDATE inventory SET is_hidden = 1 WHERE id = %s", (item_id,)) conn.commit() cursor.close() flash("Item hidden and moved to the archive list.", "warning") except Error as e: flash(f"Error hiding item: {e}", "danger") return redirect(url_for('index')) @app.route('/unhide/', methods=['POST']) def unhide_item(item_id): """Restores an archived item back to visible production lists.""" try: conn = get_db() cursor = conn.cursor() cursor.execute("UPDATE inventory SET is_hidden = 0 WHERE id = %s", (item_id,)) conn.commit() cursor.close() flash("Item restored back to active view.", "success") except Error as e: flash(f"Error restoring item: {e}", "danger") return redirect(url_for('index')) @app.route('/delete/', methods=['POST']) def delete_item(item_id): """Permanently deletes/removes the item from the database record.""" try: conn = get_db() cursor = conn.cursor() cursor.execute("DELETE FROM inventory WHERE id = %s", (item_id,)) conn.commit() cursor.close() flash("Item permanently removed.", "danger") except Error as e: flash(f"Error permanently removing item: {e}", "danger") return redirect(url_for('index')) #
